// Firestore Security Rules for Event Management App
// Navigate to Firebase Console > Firestore Database > Rules tab
// Copy this entire content and paste it into the Rules editor
// Click Publish when done

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // Users collection
    match /users/{userId} {
      // Users can read their own document
      allow read: if request.auth.uid == userId;
      
      // Admins can read all users
      allow read: if isAdmin(request.auth.uid);
      
      // Users can write their own document (but NOT change role)
      allow write: if request.auth.uid == userId 
        && request.resource.data.role == resource.data.role;
      
      // Only admins can change user roles (for promotion)
      allow update: if isAdmin(request.auth.uid) 
        && request.resource.data.role in ['admin', 'student'];
    }
    
    // Events collection
    match /events/{eventId} {
      // Check if user is admin
      function isAdmin(uid) {
        return get(/databases/$(database)/documents/users/$(uid)).data.role == 'admin';
      }
      
      // Check if approval status is valid
      function isValidApprovalStatus(status) {
        return status in ['pending', 'accepted', 'rejected'];
      }
      
      // Check if status is valid
      function isValidStatus(status) {
        return status in ['pending', 'accepted', 'rejected'];
      }
      
      // STUDENTS can read events (all events are public)
      allow read: if request.auth != null;
      
      // STUDENTS can create new events
      allow create: if request.auth != null 
        && request.resource.data.createdBy == request.auth.uid
        && isValidApprovalStatus(request.resource.data.approvalStatus)
        && isValidStatus(request.resource.data.status)
        && request.resource.data.approvalStatus == 'pending';
      
      // ADMINS can approve/reject/set-pending events
      allow update: if isAdmin(request.auth.uid)
        && isValidApprovalStatus(request.resource.data.approvalStatus)
        && isValidStatus(request.resource.data.status);
      
      // Only creators and co-hosts can update their own events
      allow update: if request.auth.uid == resource.data.createdBy 
        || request.auth.uid in resource.data.coHostIds
        && request.resource.data.approvalStatus == resource.data.approvalStatus;
      
      // Only admins can delete events
      allow delete: if isAdmin(request.auth.uid);
    }
  }
}
